cmake_minimum_required(VERSION 3.16)
project(gestalt_radtan_check LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core calib3d)
get_filename_component(OPENVINS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT OPENVINS_SOURCE)
  message(FATAL_ERROR "OPENVINS_SOURCE must be an isolated patched source tree")
endif()
add_executable(radtan_check radtan.cc)
target_include_directories(radtan_check PRIVATE "${OPENVINS_SOURCE}/ov_core/src" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(radtan_check PRIVATE Eigen3::Eigen ${OpenCV_LIBS})
enable_testing()
add_test(NAME full_radtan_projection_inverse_and_jacobians COMMAND radtan_check)
add_executable(feature_storage_check feature_storage.cc
  "${OPENVINS_SOURCE}/ov_core/src/feat/Feature.cpp"
  "${OPENVINS_SOURCE}/ov_core/src/feat/FeatureDatabase.cpp")
target_include_directories(feature_storage_check PRIVATE "${OPENVINS_SOURCE}/ov_core/src" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(feature_storage_check PRIVATE Eigen3::Eigen ${OpenCV_LIBS})
add_test(NAME feature_history_preserves_order_and_payloads COMMAND feature_storage_check)
